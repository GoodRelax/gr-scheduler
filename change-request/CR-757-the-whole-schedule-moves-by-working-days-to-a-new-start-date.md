4.6 節の働いた例は、次の Python（標準のライブラリだけ）で数えた。暦は月〜金と例外日 2029-03-20、写像は行 3〜行 5 のとおり。

```python
import datetime as dt
HOLIDAYS = {dt.date(2029, 3, 20)}
def working(d): return d.weekday() < 5 and d not in HOLIDAYS
def prev_working(d):
    d -= dt.timedelta(1)
    while not working(d): d -= dt.timedelta(1)
    return d
def rank(origin, d):
    k, step, at = 0, (1 if d >= origin else -1), origin
    while at != d:
        at += dt.timedelta(step)
        if working(at): k += step
    return k
def day_of_rank(origin, k):
    at, step, left = origin, (1 if k >= 0 else -1), abs(k)
    while left:
        at += dt.timedelta(step)
        if working(at): left -= 1
    return at
def moved(o_old, o_new, d):
    src = d if working(d) else prev_working(d)
    return day_of_rank(o_new, rank(o_old, src))
o_old, o_new = dt.date(2028, 4, 4), dt.date(2029, 3, 5)
print('working days moved', rank(o_old, o_new))   # 239
for d in [dt.date(2028, 4, 4), dt.date(2028, 4, 7), dt.date(2028, 4, 14), dt.date(2028, 5, 1), dt.date(2028, 5, 2),
          dt.date(2028, 5, 6), dt.date(2028, 4, 29), dt.date(2028, 4, 30), dt.date(2028, 4, 10), dt.date(2028, 3, 31), dt.date(2028, 4, 3)]:
    print(d, moved(o_old, o_new, d))
```
